using System.Collections;
using Mediapipe.Unity.Sample.HandLandmarkDetection;
using UnityEngine;

public class HandDetection : MonoBehaviour
{
    [Header("References")]
    [SerializeField] private HandDataReceiver receiver;
    [SerializeField] private HandLandmarkerRunner handLandmarker;

    [Header("Timing")]
    [SerializeField] private float warmupTime = 0.5f;
    [SerializeField] private float heartHoldTime = 0.25f;
    [SerializeField] private float phase2Duration = 1f;
    [SerializeField] private float phase3Duration = 1f;
    [SerializeField] private float phase4Duration = 1f;

    [Header("Scoring")]
    [SerializeField] private float phase2MoveThreshold = 0.05f; // 0.04f – 0.07f
    [SerializeField] private float phase3MoveThreshold = 0.05f; // 0.04f – 0.07f
    [SerializeField] private float phase4MoveThreshold = 0.03f; // 0.02f – 0.04f

    [Header("Heart Validation")]
    [SerializeField] private float allowedDistanceDrift = 0.15f;

    [SerializeField] private int phase;
    [SerializeField] private int point;

    private float phaseTimer;
    private float heartHoldTimer;
    private float startHandDistance;
    private float accumulatedMoveX;
    private bool isDetecting;

    public event System.Action OnHandCheckStart = delegate { };
    public event System.Action<int> OnHandDetectionProceed = delegate { };
    public event System.Action<int> OnHandDetectionOver = delegate { };

    // void Start()
    // {
    //     receiver.gameObject.SetActive(false);
    // }

    // private void Start()
    // {
    //     StartCoroutine(FinishLankmarkSetup());
    // }

    // IEnumerator FinishLankmarkSetup()
    // {
    //     yield return new WaitForSeconds(1f);
    //     StartCoroutine(StopRunnerDelayed());
    // }

    private void Update()
    {
        if (!isDetecting)
            return;

        switch (phase)
        {
            case 0:
                UpdateWarmup();
                break;

            case 1:
                UpdatePhase1();
                break;

            case 2:
                UpdateMotionPhase(phase2Duration, phase2MoveThreshold, "萌え", nextPhase: 3);
                break;

            case 3:
                UpdateMotionPhase(phase3Duration, phase3MoveThreshold, "萌え", nextPhase: 4);
                break;

            case 4:
                UpdateLastPhase();
                break;
        }
    }

    public void StartCheck()
    {
        // StopAllCoroutines();

        point = 0;
        phase = 0;
        phaseTimer = 0f;
        heartHoldTimer = 0f;
        startHandDistance = 0f;
        accumulatedMoveX = 0f;
        isDetecting = true;

        // handLandmarker.gameObject.SetActive(true);
        // receiver.gameObject.SetActive(true);
        // StartCoroutine(handLandmarker.Resume());
        // StartCoroutine(StartRunnerDelayed());

        Debug.Log("************************* Hand Check Start *************************");
    }

    public void EndCheck()
    {
        if (!isDetecting)
            return;

        isDetecting = false;

        // handLandmarker.gameObject.SetActive(false);
        // receiver.gameObject.SetActive(false);
        // handLandmarker.Pause();
        // StartCoroutine(StopRunnerDelayed());

        Debug.Log($"************************* Hand Check End / Point: {point} *************************");
        OnHandDetectionOver?.Invoke(point);
    }

    private IEnumerator StartRunnerDelayed()
    {
        // wait one frame for Unity lifecycle to initialize
        yield return null;

        yield return handLandmarker.Resume();

        Debug.Log("HandLandmarker resumed");
    }

    private IEnumerator StopRunnerDelayed()
    {
        // Stop processing first
        handLandmarker.Pause();
        yield return null;

        // Then disable
        // handLandmarker.gameObject.SetActive(false);
        receiver.gameObject.SetActive(false);

        Debug.Log("HandLandmarker stopped");
    }

    private void UpdateWarmup()
    {
        phaseTimer += Time.deltaTime;

        if (phaseTimer >= warmupTime)
        {
            phase = 1;
            phaseTimer = 0f;
            heartHoldTimer = 0f;
        }
    }

    private void UpdatePhase1()
    {
        bool hasHands = receiver.Result.handLandmarks.Count >= 1;
        // bool stable = receiver.AreHandsStable(2);

        if (!hasHands)
        {
            Debug.Log("Hands not present.");
            heartHoldTimer = 0f;
            return;
        }

        // if (!receiver.AreTwoHandsPresent())
        // if (!receiver.AreTwoHandsPresentRelaxed())
        // {
        //     Debug.Log("Hands not present.");
        //     heartHoldTimer = 0f;
        //     return;
        // }

        Debug.Log("Hands present.");

        bool heartPose = receiver.IsHeartPose();
        bool stable = receiver.AreHandsStable(2);

        if (heartPose && stable)
        {
            heartHoldTimer += Time.deltaTime;

            if (heartHoldTimer >= heartHoldTime)
            {
                startHandDistance = receiver.HandsDistance;
                phase = 2;
                phaseTimer = 0f;
                accumulatedMoveX = 0f;

                Debug.Log("Heart pose accepted. Start sequence.");
                OnHandCheckStart?.Invoke();
                OnHandDetectionProceed?.Invoke(phase);
            }
        }
        else
        {
            heartHoldTimer = 0f;
        }
    }

    private void UpdateMotionPhase(float duration, float moveThreshold, string debugText, int nextPhase)
    {
        phaseTimer += Time.deltaTime;

        if (receiver.AreTwoHandsPresent())
        {
            accumulatedMoveX += receiver.GetAverageAbsDeltaX();
        }

        Debug.Log($"MoveX: {accumulatedMoveX}");

        if (phaseTimer >= duration)
        {
            if (receiver.AreTwoHandsPresent() &&
                Mathf.Abs(receiver.HandsDistance - startHandDistance) <= allowedDistanceDrift &&
                accumulatedMoveX >= moveThreshold)
            {
                point++;
            }

            Debug.Log(debugText);

            phase = nextPhase;
            phaseTimer = 0f;
            accumulatedMoveX = 0f;

            OnHandDetectionProceed?.Invoke(phase);
        }
    }

    private void UpdateLastPhase()
    {
        phaseTimer += Time.deltaTime;

        if (receiver.AreTwoHandsPresent())
        {
            accumulatedMoveX += receiver.GetAverageAbsDeltaX();
        }

        if (phaseTimer >= phase4Duration)
        {
            if (receiver.AreTwoHandsPresent() &&
                Mathf.Abs(receiver.HandsDistance - startHandDistance) <= allowedDistanceDrift &&
                accumulatedMoveX >= phase4MoveThreshold)
            {
                point++;
            }

            Debug.Log("キュン");
            EndCheck();
        }
    }
}
